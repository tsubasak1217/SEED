# ===========================================================
#  わらしべフィッシング / 釣りバトルのビートパターン
# -----------------------------------------------------------
#  1 行 = 戦闘サイクル 1 周ぶん（既定 2 小節・4 拍子）
#
#   {1/n}  … 以降のカンマ 1 つが 1/n 音符（小節を 1.0 とした長さ）
#   ,      … その音価で 1 単位進める（カンマの合計が 1.0 で 1 小節）
#   t      … 直後のカンマの先頭で叩く
#   [...]  … 対象の魚レベル帯（[1-3] [1-3,5] [5] [3-] [-2] [*]）
#   #      … 以降は行末までコメント／空行は無視
#
#  記法の詳細は docs/beat_patterns.md を参照。
#  このファイルを保存すると、次の戦闘から自動で反映される（ホットリロード）。
# ===========================================================

{1/4}t,,,,t,,,,                               [-2]  # 1拍目だけ（最初の 1 匹用）
{1/4}t,,t,,t,,t,,                             [1-2]  # 1拍目と3拍目
{1/4}t,t,,,t,t,,,                             [1-2]  # 頭 2 拍を続けて叩く
{1/4}t,,,t,t,,,t,                             [2-4]  # 1拍目と4拍目
{1/4}t,,t,,{1/8}t,,t,,t,,,,                   [3-5]  # 2小節目だけ 8 分
{1/6}t,,t,,t,,{1/4}t,,t,,                     [3-5]  # 3連符→4分
{1/6}t,t,,{1/4}t,,t,,t,,                      [1-3,5]  # 1小節内で 1/6 と 1/4 を混在
{1/8}t,,t,t,,,t,,{1/6}t,,t,,t,,               [5]  # 8分→3連符
{1/6}t,t,,t,t,,{1/8}t,,t,,t,,t,,              [5-7]  # 3連符→8分
{1/8}t,t,,t,t,,t,,{1/6}t,,t,t,,t,             [6-]  # 裏拍を含む 8 分
{1/8}t,,t,,t,t,,t,{1/12}t,,,t,,,t,,,t,,,      [8-]  # 16 分相当（1/12）を含む最上位
{1/4}t,,,,t,,,,                               [*]  # どのレベルにも該当しないときの保険
